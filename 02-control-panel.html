01 / LIVE PATCH

140 ms

Edit a script, save, and the running game picks it up. No restart, no lost state, no reloading the level you spent four minutes reaching.

// player.scrap — edited while the game runs
fn update(p: &mut Player, input: &Input, dt: f32) {
    let dir = input.axis("move");
    p.vel += dir * ACCEL * dt;
-   p.vel *= 0.90;
+   p.vel *= 0.82;   // snappier stop
    if input.pressed("jump") && p.grounded {
        p.vel.y = JUMP_SPEED;
        sfx::play("clank.wav");
    }
}
